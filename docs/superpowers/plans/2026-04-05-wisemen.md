# Wisemen Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a web-based multi-model chat interface (Wisemen) on Cloudflare, porting Think Tank's pipeline logic to a SvelteKit app with streaming, persistent conversations, and profiles.

**Architecture:** SvelteKit frontend with server routes running as Cloudflare Workers. D1 (SQLite) for conversation/profile storage. SSE streaming for real-time model responses. Cloudflare Access for auth. Sentry for monitoring.

**Tech Stack:** SvelteKit, TypeScript, Cloudflare Pages/Workers/D1, SSE, Sentry (`@sentry/sveltekit`, `@sentry/cloudflare`)

**Spec:** `docs/superpowers/specs/2026-04-05-wisemen-design.md` (in the think-tank repo)

CREATE TABLE IF NOT EXISTS profiles (
    id          TEXT PRIMARY KEY,
    name        TEXT NOT NULL,
    context     TEXT DEFAULT '',
    created_at  INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS conversations (
    id          TEXT PRIMARY KEY,
    profile_id  TEXT NOT NULL REFERENCES profiles(id),
    title       TEXT,
    created_at  INTEGER NOT NULL,
    updated_at  INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS messages (
    id          TEXT PRIMARY KEY,
    conv_id     TEXT NOT NULL REFERENCES conversations(id),
    role        TEXT NOT NULL,
    model       TEXT,
    content     TEXT NOT NULL,
    elapsed     REAL,
    turn        INTEGER NOT NULL,
    created_at  INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS syntheses (
    id          TEXT PRIMARY KEY,
    conv_id     TEXT NOT NULL REFERENCES conversations(id),
    analysis    TEXT,
    rankings    TEXT,
    label_map   TEXT,
    synthesis   TEXT,
    chairman    TEXT DEFAULT 'claude',
    turn        INTEGER,
    created_at  INTEGER NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_messages_conv ON messages(conv_id, turn);
CREATE INDEX IF NOT EXISTS idx_conversations_profile ON conversations(profile_id, updated_at DESC);

import type { D1Database } from '@cloudflare/workers-types';
import type { Profile, Conversation, Message, Synthesis, ConversationWithMeta } from './types';

// ── Profiles ──

export async function listProfiles(db: D1Database): Promise<Profile[]> {
	const result = await db.prepare('SELECT * FROM profiles ORDER BY created_at').all<Profile>();
	return result.results;
}

export async function getProfile(db: D1Database, id: string): Promise<Profile | null> {
	return db.prepare('SELECT * FROM profiles WHERE id = ?').bind(id).first<Profile>();
}

export async function createProfile(db: D1Database, id: string, name: string, context = ''): Promise<void> {
	await db.prepare('INSERT INTO profiles (id, name, context, created_at) VALUES (?, ?, ?, ?)')
		.bind(id, name, context, Date.now()).run();
}

export async function updateProfile(db: D1Database, id: string, name: string, context: string): Promise<void> {
	await db.prepare('UPDATE profiles SET name = ?, context = ? WHERE id = ?')
		.bind(name, context, id).run();
}

// ── Conversations ──

export async function listConversations(db: D1Database, profileId: string): Promise<ConversationWithMeta[]> {
	const result = await db.prepare(`
		SELECT c.*,
			COALESCE(MAX(m.turn), 0) AS turn_count,
			CASE WHEN EXISTS (SELECT 1 FROM syntheses s WHERE s.conv_id = c.id) THEN 1 ELSE 0 END AS has_synthesis
		FROM conversations c
		LEFT JOIN messages m ON m.conv_id = c.id
		WHERE c.profile_id = ?
		GROUP BY c.id
		ORDER BY c.updated_at DESC
	`).bind(profileId).all<ConversationWithMeta>();
	return result.results;
}

export async function getConversation(db: D1Database, id: string): Promise<Conversation | null> {
	return db.prepare('SELECT * FROM conversations WHERE id = ?').bind(id).first<Conversation>();
}

export async function createConversation(db: D1Database, id: string, profileId: string, title: string | null): Promise<void> {
	const now = Date.now();
	await db.prepare('INSERT INTO conversations (id, profile_id, title, created_at, updated_at) VALUES (?, ?, ?, ?, ?)')
		.bind(id, profileId, title, now, now).run();
}

export async function updateConversationTitle(db: D1Database, id: string, title: string): Promise<void> {
	await db.prepare('UPDATE conversations SET title = ?, updated_at = ? WHERE id = ?')
		.bind(title, Date.now(), id).run();
}

export async function deleteConversation(db: D1Database, id: string): Promise<void> {
	await db.prepare('DELETE FROM messages WHERE conv_id = ?').bind(id).run();
	await db.prepare('DELETE FROM syntheses WHERE conv_id = ?').bind(id).run();
	await db.prepare('DELETE FROM conversations WHERE id = ?').bind(id).run();
}

export async function touchConversation(db: D1Database, id: string): Promise<void> {
	await db.prepare('UPDATE conversations SET updated_at = ? WHERE id = ?').bind(Date.now(), id).run();
}

// ── Messages ──

export async function getMessages(db: D1Database, convId: string): Promise<Message[]> {
	const result = await db.prepare('SELECT * FROM messages WHERE conv_id = ? ORDER BY turn, created_at')
		.bind(convId).all<Message>();
	return result.results;
}

export async function getMaxTurn(db: D1Database, convId: string): Promise<number> {
	const row = await db.prepare('SELECT COALESCE(MAX(turn), 0) AS max_turn FROM messages WHERE conv_id = ?')
		.bind(convId).first<{ max_turn: number }>();
	return row?.max_turn ?? 0;
}

export async function insertMessage(
	db: D1Database, id: string, convId: string,
	role: 'user' | 'assistant', model: string | null,
	content: string, elapsed: number | null, turn: number
): Promise<void> {
	await db.prepare(
		'INSERT INTO messages (id, conv_id, role, model, content, elapsed, turn, created_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?)'
	).bind(id, convId, role, model, content, elapsed, turn, Date.now()).run();
}

// ── Syntheses ──

export async function getSyntheses(db: D1Database, convId: string): Promise<Synthesis[]> {
	const result = await db.prepare('SELECT * FROM syntheses WHERE conv_id = ? ORDER BY created_at')
		.bind(convId).all<Synthesis>();
	return result.results;
}

export async function insertSynthesis(
	db: D1Database, id: string, convId: string,
	analysis: string | null, rankings: string | null, labelMap: string | null,
	synthesis: string | null, chairman: string, turn: number
): Promise<void> {
	await db.prepare(
		'INSERT INTO syntheses (id, conv_id, analysis, rankings, label_map, synthesis, chairman, turn, created_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)'
	).bind(id, convId, analysis, rankings, labelMap, synthesis, chairman, turn, Date.now()).run();
}

import { MODELS } from './config';

export type ChatMessage = { role: 'system' | 'user' | 'assistant'; content: string };

export interface StreamCallbacks {
	onToken: (text: string) => void;
	onDone: (fullText: string, elapsed: number) => void;
	onError: (error: string) => void;
}

export async function callClaudeStream(
	messages: ChatMessage[], apiKey: string, cb: StreamCallbacks
): Promise<void> {
	const systemParts = messages.filter(m => m.role === 'system').map(m => m.content);
	const apiMessages = messages.filter(m => m.role !== 'system');
	const start = Date.now();

	const resp = await fetch('https://api.anthropic.com/v1/messages', {
		method: 'POST',
		headers: {
			'x-api-key': apiKey,
			'anthropic-version': '2023-06-01',
			'content-type': 'application/json',
		},
		body: JSON.stringify({
			model: MODELS.claude.modelId,
			max_tokens: 4096,
			stream: true,
			system: systemParts.join('\n\n'),
			messages: apiMessages,
		}),
	});

	if (!resp.ok) {
		const body = await resp.text();
		cb.onError(`Claude API error ${resp.status}: ${body.slice(0, 200)}`);
		return;
	}

	let fullText = '';
	const reader = resp.body!.getReader();
	const decoder = new TextDecoder();
	let buffer = '';

	while (true) {
		const { done, value } = await reader.read();
		if (done) break;
		buffer += decoder.decode(value, { stream: true });

		const lines = buffer.split('\n');
		buffer = lines.pop()!;

		for (const line of lines) {
			if (!line.startsWith('data: ')) continue;
			const data = line.slice(6);
			if (data === '[DONE]') continue;
			try {
				const event = JSON.parse(data);
				if (event.type === 'content_block_delta' && event.delta?.text) {
					fullText += event.delta.text;
					cb.onToken(event.delta.text);
				}
			} catch { /* skip non-JSON lines */ }
		}
	}

	cb.onDone(fullText, (Date.now() - start) / 1000);
}

export async function callOpenAICompatStream(
	messages: ChatMessage[], apiKey: string, baseUrl: string, modelId: string, cb: StreamCallbacks
): Promise<void> {
	const start = Date.now();

	const resp = await fetch(`${baseUrl}/chat/completions`, {
		method: 'POST',
		headers: {
			'Authorization': `Bearer ${apiKey}`,
			'Content-Type': 'application/json',
		},
		body: JSON.stringify({
			model: modelId,
			max_completion_tokens: 4096,
			stream: true,
			messages,
		}),
	});

	if (!resp.ok) {
		const body = await resp.text();
		cb.onError(`API error ${resp.status}: ${body.slice(0, 200)}`);
		return;
	}

	let fullText = '';
	const reader = resp.body!.getReader();
	const decoder = new TextDecoder();
	let buffer = '';

	while (true) {
		const { done, value } = await reader.read();
		if (done) break;
		buffer += decoder.decode(value, { stream: true });

		const lines = buffer.split('\n');
		buffer = lines.pop()!;

		for (const line of lines) {
			if (!line.startsWith('data: ')) continue;
			const data = line.slice(6);
			if (data === '[DONE]') continue;
			try {
				const event = JSON.parse(data);
				const content = event.choices?.[0]?.delta?.content;
				if (content) {
					fullText += content;
					cb.onToken(content);
				}
			} catch { /* skip */ }
		}
	}

	cb.onDone(fullText, (Date.now() - start) / 1000);
}

export async function callGeminiStream(
	messages: ChatMessage[], apiKey: string, cb: StreamCallbacks
): Promise<void> {
	const start = Date.now();
	let systemText = '';
	const contents: Array<{ role: string; parts: Array<{ text: string }> }> = [];

	for (const m of messages) {
		if (m.role === 'system') {
			systemText += m.content + '\n\n';
		} else if (m.role === 'user') {
			contents.push({ role: 'user', parts: [{ text: m.content }] });
		} else {
			contents.push({ role: 'model', parts: [{ text: m.content }] });
		}
	}

	const body: Record<string, unknown> = {
		contents,
		generationConfig: { maxOutputTokens: 4096 },
	};
	if (systemText.trim()) {
		body.systemInstruction = { parts: [{ text: systemText.trim() }] };
	}

	const url = `https://generativelanguage.googleapis.com/v1beta/models/${MODELS.gemini.modelId}:streamGenerateContent?alt=sse&key=${apiKey}`;

	const resp = await fetch(url, {
		method: 'POST',
		headers: { 'Content-Type': 'application/json' },
		body: JSON.stringify(body),
	});

	if (!resp.ok) {
		const respBody = await resp.text();
		cb.onError(`Gemini API error ${resp.status}: ${respBody.slice(0, 200)}`);
		return;
	}

	let fullText = '';
	const reader = resp.body!.getReader();
	const decoder = new TextDecoder();
	let buffer = '';

	while (true) {
		const { done, value } = await reader.read();
		if (done) break;
		buffer += decoder.decode(value, { stream: true });

		const lines = buffer.split('\n');
		buffer = lines.pop()!;

		for (const line of lines) {
			if (!line.startsWith('data: ')) continue;
			try {
				const event = JSON.parse(line.slice(6));
				const text = event.candidates?.[0]?.content?.parts?.[0]?.text;
				if (text) {
					fullText += text;
					cb.onToken(text);
				}
			} catch { /* skip */ }
		}
	}

	cb.onDone(fullText, (Date.now() - start) / 1000);
}

export function callGptStream(messages: ChatMessage[], apiKey: string, cb: StreamCallbacks) {
	return callOpenAICompatStream(messages, apiKey, 'https://api.openai.com/v1', MODELS.gpt.modelId, cb);
}

import type { Message } from '$lib/db/types';
import type { ChatMessage } from '$lib/models/callers';
import { MODELS } from '$lib/models/config';
import { fillTemplate, SYSTEM_PROMPT, DELIBERATION_PROMPT } from './prompts';

export function buildModelMessages(
	allMessages: Message[],
	modelKey: string,
	profileContext: string,
	activeModels: string[]
): ChatMessage[] {
	const result: ChatMessage[] = [];

	// System message
	const systemContent = fillTemplate(SYSTEM_PROMPT, { modelCount: activeModels.length });
	const systemParts = [systemContent];
	if (profileContext.trim()) {
		systemParts.push(`User context:\n${profileContext}`);
	}
	result.push({ role: 'system', content: systemParts.join('\n\n') });

	// Group messages by turn
	const turns = new Map<number, Message[]>();
	for (const msg of allMessages) {
		const group = turns.get(msg.turn) || [];
		group.push(msg);
		turns.set(msg.turn, group);
	}

	const sortedTurns = [...turns.keys()].sort((a, b) => a - b);

	for (const turn of sortedTurns) {
		const turnMsgs = turns.get(turn)!;
		const userMsg = turnMsgs.find(m => m.role === 'user');
		const modelResponse = turnMsgs.find(m => m.role === 'assistant' && m.model === modelKey);
		const otherResponses = turnMsgs.filter(
			m => m.role === 'assistant' && m.model !== modelKey && m.model !== null
		);

		// For turn > 1, prepend other models' responses as deliberation context
		if (turn > 1 && otherResponses.length > 0) {
			const otherParts = otherResponses
				.map(m => `**${MODELS[m.model!]?.name ?? m.model}:**\n${m.content}`)
				.join('\n\n---\n\n');

			const deliberation = fillTemplate(DELIBERATION_PROMPT, { otherResponses: otherParts });

			if (userMsg) {
				result.push({ role: 'user', content: `${deliberation}\n\n---\n\nThe user has followed up:\n\n${userMsg.content}` });
			} else {
				result.push({ role: 'user', content: deliberation });
			}
		} else if (userMsg) {
			result.push({ role: 'user', content: userMsg.content });
		}

		if (modelResponse) {
			result.push({ role: 'assistant', content: modelResponse.content });
		}
	}

	return result;
}

export interface AnonymizedResponses {
	anonymized: Record<string, string>;  // "Response A" -> text
	labelToModel: Record<string, string>; // "Response A" -> "gpt"
}

export function anonymizeResponses(
	responses: Record<string, string>,
	activeModels: string[]
): AnonymizedResponses {
	const responding = activeModels.filter(k => responses[k]);
	const labels = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'.slice(0, responding.length).split('');

	// Shuffle
	const shuffled = [...responding];
	for (let i = shuffled.length - 1; i > 0; i--) {
		const j = Math.floor(Math.random() * (i + 1));
		[shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
	}

	const anonymized: Record<string, string> = {};
	const labelToModel: Record<string, string> = {};

	for (let i = 0; i < shuffled.length; i++) {
		const label = `Response ${labels[i]}`;
		anonymized[label] = responses[shuffled[i]];
		labelToModel[label] = shuffled[i];
	}

	return { anonymized, labelToModel };
}

export function parseRanking(text: string): string[] {
	if (!text.includes('FINAL RANKING:')) {
		return [...text.matchAll(/Response [A-Z]/g)].map(m => m[0]);
	}

	const rankingSection = text.split('FINAL RANKING:')[1];
	const numbered = [...rankingSection.matchAll(/\d+\.\s*Response [A-Z]/g)];
	if (numbered.length > 0) {
		return numbered.map(m => m[0].match(/Response [A-Z]/)![0]);
	}

	return [...rankingSection.matchAll(/Response [A-Z]/g)].map(m => m[0]);
}

export interface AggregateRanking {
	model: string;
	avgRank: number;
	votes: number;
}

export function calculateAggregateRankings(
	allRankings: Array<{ reviewer: string; ranking: string[] }>,
	labelToModel: Record<string, string>
): AggregateRanking[] {
	const positions: Record<string, number[]> = {};

	for (const { ranking } of allRankings) {
		for (let i = 0; i < ranking.length; i++) {
			const label = ranking[i];
			if (label in labelToModel) {
				const modelKey = labelToModel[label];
				if (!positions[modelKey]) positions[modelKey] = [];
				positions[modelKey].push(i + 1);
			}
		}
	}

	const aggregate: AggregateRanking[] = [];
	for (const [model, posList] of Object.entries(positions)) {
		const avg = posList.reduce((a, b) => a + b, 0) / posList.length;
		aggregate.push({ model, avgRank: Math.round(avg * 100) / 100, votes: posList.length });
	}

	aggregate.sort((a, b) => a.avgRank - b.avgRank);
	return aggregate;
}

import { MODELS, DEFAULT_CHAIRMAN } from '$lib/models/config';
import { resolveModel } from '$lib/models/resolver';
import type { ChatMessage } from '$lib/models/callers';
import { anonymizeResponses } from './anonymizer';
import { parseRanking, calculateAggregateRankings, type AggregateRanking } from './ranking';
import { fillTemplate, REVIEW_PROMPT, ANALYSIS_PROMPT, CHAIRMAN_PROMPT } from './prompts';

export interface SynthesisResult {
	analysis: string | null;
	rankings: AggregateRanking[];
	labelMap: Record<string, string>;
	synthesis: string | null;
}

export interface SynthesisCallbacks {
	onPhase: (phase: string, status: string) => void;
	onReviewComplete: (model: string, ranking: string[]) => void;
	onAnalysisToken: (text: string) => void;
	onSynthesisToken: (text: string) => void;
	onError: (phase: string, model: string, error: string) => void;
}

export async function orchestrateSynthesis(
	responses: Record<string, string>,
	activeModels: string[],
	question: string,
	env: Record<string, string>,
	cb: SynthesisCallbacks,
	chairmanKey = DEFAULT_CHAIRMAN
): Promise<SynthesisResult> {
	const { anonymized, labelToModel } = anonymizeResponses(responses, activeModels);
	const modelCount = Object.keys(responses).length;

	// ── Phase 1: Peer Review (parallel) ──
	cb.onPhase('review', 'started');

	const responsesText = Object.entries(anonymized)
		.map(([label, text]) => `${label}:\n${text}`)
		.join('\n\n---\n\n');

	const rankingSlots = Object.keys(anonymized)
		.map((_, i) => `${i + 1}. Response X`)
		.join('\n');

	const reviewPrompt = fillTemplate(REVIEW_PROMPT, { question, responsesText, rankingSlots });
	const reviewMessages: ChatMessage[] = [
		{ role: 'system', content: 'You are a careful, objective evaluator.' },
		{ role: 'user', content: reviewPrompt },
	];

	const allRankings: Array<{ reviewer: string; ranking: string[] }> = [];

	const reviewPromises = activeModels.map(async (key) => {
		const resolved = resolveModel(key, env);
		if (!resolved) return;

		let fullText = '';
		try {
			await resolved.caller(reviewMessages, {
				onToken: () => {},  // review isn't streamed to client
				onDone: (text) => {
					fullText = text;
					const ranking = parseRanking(text);
					allRankings.push({ reviewer: key, ranking });
					cb.onReviewComplete(key, ranking);
				},
				onError: (err) => cb.onError('review', key, err),
			});
		} catch (e) {
			cb.onError('review', key, String(e));
		}
	});

	await Promise.all(reviewPromises);
	const rankings = calculateAggregateRankings(allRankings, labelToModel);

	// ── Phase 2: Analysis (chairman, streamed) ──
	cb.onPhase('analysis', 'started');

	const analysisResponsesText = activeModels
		.filter(k => responses[k])
		.map(k => `**${MODELS[k].name}:**\n${responses[k]}`)
		.join('\n\n---\n\n');

	const analysisPrompt = fillTemplate(ANALYSIS_PROMPT, {
		modelCount, question, responsesText: analysisResponsesText,
	});

	const analysisMessages: ChatMessage[] = [
		{ role: 'system', content: 'You are an expert analyst. Extract the structure of agreement and disagreement from multi-model responses. Be precise and specific.' },
		{ role: 'user', content: analysisPrompt },
	];

	let analysisText: string | null = null;
	const chairmanResolved = resolveModel(chairmanKey, env);

	if (chairmanResolved) {
		try {
			await chairmanResolved.caller(analysisMessages, {
				onToken: (text) => cb.onAnalysisToken(text),
				onDone: (text) => { analysisText = text; },
				onError: (err) => cb.onError('analysis', chairmanKey, err),
			});
		} catch (e) {
			cb.onError('analysis', chairmanKey, String(e));
		}
	}

	// ── Phase 3: Chairman Synthesis (streamed) ──
	cb.onPhase('synthesis', 'started');

	const rankingsText = rankings
		.map((r, i) => `  ${i + 1}. ${MODELS[r.model]?.name ?? r.model} — avg rank ${r.avgRank}`)
		.join('\n');

	const chairmanPrompt = fillTemplate(CHAIRMAN_PROMPT, {
		modelCount, question, responsesText: analysisResponsesText,
		rankingsText, analysisText: analysisText ?? 'Analysis not available.',
	});

	const chairmanMessages: ChatMessage[] = [
		{ role: 'system', content: 'You are the chairman of a multi-model think tank. Synthesize the council\'s work into a single authoritative answer.' },
		{ role: 'user', content: chairmanPrompt },
	];

	let synthesisText: string | null = null;

	if (chairmanResolved) {
		try {
			await chairmanResolved.caller(chairmanMessages, {
				onToken: (text) => cb.onSynthesisToken(text),
				onDone: (text) => { synthesisText = text; },
				onError: (err) => cb.onError('synthesis', chairmanKey, err),
			});
		} catch (e) {
			cb.onError('synthesis', chairmanKey, String(e));
		}
	}

	return { analysis: analysisText, rankings, labelMap: labelToModel, synthesis: synthesisText };
}

export interface SSEOptions {
	onEvent: (event: string, data: unknown) => void;
	onError?: (error: Event) => void;
	onClose?: () => void;
}

export function streamFetch(url: string, body: unknown, options: SSEOptions): AbortController {
	const controller = new AbortController();

	(async () => {
		try {
			const resp = await fetch(url, {
				method: 'POST',
				headers: { 'Content-Type': 'application/json' },
				body: JSON.stringify(body),
				signal: controller.signal,
			});

			if (!resp.ok) {
				const text = await resp.text();
				options.onEvent('error', { error: `HTTP ${resp.status}: ${text}` });
				return;
			}

			const reader = resp.body!.getReader();
			const decoder = new TextDecoder();
			let buffer = '';

			while (true) {
				const { done, value } = await reader.read();
				if (done) break;
				buffer += decoder.decode(value, { stream: true });

				const segments = buffer.split('\n\n');
				buffer = segments.pop()!;

				for (const segment of segments) {
					let eventType = 'message';
					let data = '';

					for (const line of segment.split('\n')) {
						if (line.startsWith('event: ')) eventType = line.slice(7);
						else if (line.startsWith('data: ')) data = line.slice(6);
					}

					if (data) {
						try {
							options.onEvent(eventType, JSON.parse(data));
						} catch {
							options.onEvent(eventType, data);
						}
					}
				}
			}

			options.onClose?.();
		} catch (e) {
			if ((e as Error).name !== 'AbortError') {
				options.onError?.(e as Event);
			}
		}
	})();

	return controller;
}

export function parseCommand(input: string): ParsedCommand | null {
	const trimmed = input.trim();
	if (!trimmed.startsWith('/')) return null;

	const spaceIndex = trimmed.indexOf(' ');
	if (spaceIndex === -1) {

export function getDateGroup(timestamp: number): string {
	const now = new Date();
	const date = new Date(timestamp);

	const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
	const yesterday = new Date(today.getTime() - 86400000);

	if (date >= today) return 'Today';
	if (date >= yesterday) return 'Yesterday';
	return 'Older';
}

export function groupByDate<T extends { updated_at: number }>(items: T[]): Map<string, T[]> {
	const groups = new Map<string, T[]>();
	for (const item of items) {
		const group = getDateGroup(item.updated_at);
		if (!groups.has(group)) groups.set(group, []);
		groups.get(group)!.push(item);
	}
	return groups;
}

				onSynthesize();
				return;
			}
			if (cmd.command === 'review') {
				onSynthesize(); // TODO: Task 15 will add review-only mode
				return;
			}
			if (cmd.command === 'models' && cmd.args) {
				const newModels = cmd.args.split(',').map(s => s.trim()).filter(k => k in MODELS);
				if (newModels.length > 0) activeModels.set(newModels);
				return;
			}
			if (cmd.command === 'blind') {
				// TODO: Task 15 will add blind mode toggle
				return;
			}
			if (cmd.command === 'context') {
				// TODO: Task 15 will add inline context editing
				return;
			}
		}

		input = '';
		isSending.set(true);
		resetStreaming();

		// Initialize streaming state for each model
		const status: Record<string, 'streaming'> = {};
		for (const key of $activeModels) status[key] = 'streaming';
		streamingStatus.set(status);

		let convId = $activeConversationId;
		if (!convId) {
			convId = crypto.randomUUID();
			activeConversationId.set(convId);
		}

		// Placeholder turn — will be replaced by server's real turn on all-done
		let pendingTurn = -1;

		// Add user message to local store FIRST (before SSE starts)
		const userMsgId = crypto.randomUUID();
		messages.update(msgs => [...msgs, {
			id: userMsgId,
			conv_id: convId!,
			role: 'user' as const,
			model: null,
			content: text,
			elapsed: null,
			turn: pendingTurn,
			created_at: Date.now(),
		}]);

		streamFetch('/api/chat', {
			message: text,
			conversationId: convId,
			profileId: $activeProfileId,
			activeModels: $activeModels,
		}, {
			onEvent(event, data: any) {
				if (event === 'token') {
					streamingTokens.update(t => ({ ...t, [data.model]: (t[data.model] ?? '') + data.content }));
				} else if (event === 'done') {
					streamingStatus.update(s => ({ ...s, [data.model]: 'done' }));
					streamingElapsed.update(e => ({ ...e, [data.model]: data.elapsed }));
					// Add completed message to store with pending turn
					messages.update(msgs => [...msgs, {
						id: crypto.randomUUID(),
						conv_id: convId!,
						role: 'assistant' as const,
						model: data.model,
						content: data.full_text,
						elapsed: data.elapsed,
						turn: pendingTurn,
						created_at: Date.now(),
					}]);
				} else if (event === 'error') {
					streamingStatus.update(s => ({ ...s, [data.model]: 'error' }));
					streamingTokens.update(t => ({ ...t, [data.model]: `Error: ${data.error}` }));
				} else if (event === 'all-done') {
					// Update all pending messages with the real turn number from the server
					const realTurn = data.turn;
					messages.update(msgs => msgs.map(m =>
						m.turn === pendingTurn ? { ...m, turn: realTurn } : m
					));
					isSending.set(false);
					resetStreaming();
				}
			},
			onClose() {
				isSending.set(false);
			},
		});
	}

	function handleKeydown(e: KeyboardEvent) {
		if (e.key === 'Enter' && !e.shiftKey) {
			e.preventDefault();
			send();
		}
	}
</script>
